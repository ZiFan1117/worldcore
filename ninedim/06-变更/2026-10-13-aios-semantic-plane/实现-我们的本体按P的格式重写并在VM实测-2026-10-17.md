```python
def build_aios_ontology() -> Ontology:
    o = Ontology("aios_world", "AIOS 世界本体", "1.0")
    o.register_object_type(ObjectType(
        type_id="surface", display_name="屏面", primary_key="name",
        metadata={"wc_id": "wc:ont00000004", "instance_mode": "many"},
        properties={
            "name": Property("name", "name", PropertyType.STRING, required=True, is_primary_key=True),
            "visible": Property("visible", "visible", PropertyType.BOOLEAN),
        },
    ))
    o.register_link_type(LinkType(..., source_type="device", target_type="human", cardinality=Cardinality.ONE_TO_MANY))
    o.register_action_type(ActionType(action_id="notice.mute", parameters=[...],
        validation_rules=[], effects=[], required_roles=["agent","human"]))
    return o
```
