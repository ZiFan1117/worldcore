## §七 复现命令（照抄可用）

    # VM 侧
    cd /root/aios
    for p in aios-cognitive aios-ontology aios-action aios-policy aios-capability-runtime; do
      echo "=== BEGIN $p $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /root/tc-run.log
      timeout -k 30 7200 cargo test -p "$p" --offline > "/root/tc-$p.log" 2>&1
      echo "=== EXIT $p rc=$? $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> /root/tc-run.log
    done

    # 取原始 test result 行
    grep -n "^test result:" /root/tc-<crate>.log
