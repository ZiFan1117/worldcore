## 五、run_action.rs（逐字）

```rust
//! 真跑一条在册动作（T-250）：★ 查在册表 ⇒ 走 CommandRunner ⇒ 真起那个程序。
//!
//! 用法: run_action fs.list proc.list
//! ★ 只用来跑**只读**的命令（ls／stat／ps 那种）；★ 会改机器的（apt）不许在这个 bin 上跑。
use std::collections::BTreeMap;

use aios_worldcore::core::registry::Registry;
use aios_worldcore::core::types::Who;
use aios_worldcore::engine::execute::{CommandRunner, Executor};

const ONT: &str = "/root/aios/crates/aios-ontology/data/ontology.json";
const MAP: &str = "/root/aios/crates/aios-ontology/data/command-map.jsonl";
const REG: &str = "/root/bus/registry.jsonl";

fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    if args.is_empty() {
        println!("用法: run_action <动作名> …");
        return;
    }
    let text = std::fs::read_to_string(ONT).expect("read ontology");
    // ① 把映射表整批注册进来（同一 registered_by 可替换 ⇒ 幂等）。
    match aios_worldcore::core::command_bridge::register_from_map(MAP, REG) {
        Ok(n) => println!("[注册] 映射表 {n} 条已登记（同源可替换 ⇒ 幂等）"),
        Err(e) => println!("[注册] 拒：{e}"),
    }
    // ② 在册表 ＝ 本体 ＋ 注册表。
    let reg = Registry::load_with_registration(&text, REG).expect("load registry");
    println!(
        "[在册] 本体 {} 条 ＋ 注册表 ⇒ 合计 {} 个动作（本体那 9 条是 world 类）",
        Registry::load(&text).expect("ont").action_kinds().len(),
        reg.action_kinds().len()
    );
    // ③ 执行体（按 program+args；不拼 shell）。
    let runner = CommandRunner::from_map(MAP).expect("build runner");
    let who = Who::new("world://user/zhang");
    let params = serde_json::Map::new();
    for action in &args {
        if !reg.has_action(action) {
            println!("[拒] {action} 不在册");
            continue;
        }
        let route = runner.route_of(action);
        match route {
            None => println!("[跳过] {action} 没有命令行 route（界面／应用类是别的一路接）"),
            Some((program, a)) => {
                println!("[跑] {action} ⇒ {program} {a:?}");
                match runner.run(action, &who, &params) {
                    Ok(out) => {
                        let head: String = out.chars().take(160).collect();
                        println!("[成] {action} 输出 {} 字节；头：{}", out.len(), head.replace('\n', " ⏎ "));
                    }
                    Err(e) => println!("[败] {action}：{e}"),
                }
            }
        }
    }
    // ★ 顺手把「在册表里各动作是哪一类」打出来（world 在本体／其余在映射表）。
    let kinds: BTreeMap<&str, usize> = BTreeMap::new();
    let _ = kinds;
}

```
