## 六 · install_actions.rs（★ 安装时跑一次）（逐字）

```rust
//! 【安装时】注册一次（T-258）：★ 装软件/装执行体的时候跑它一次 —— 之后本体里就一直留着。
//!
//! ★★★ 作者口径（逐字整理）：
//!   「**这个不是别的进程和软件起，它不需要每次在启动的时候往本体里头注册，
//!     是在安装的时候一次性注册就可以了。然后这个本体里头一直会有相关的内容，
//!     卸载的时候在删除**」。
//!
//! ★ 所以：★ 注册**只在安装时发生**；★ 启动时**不注册**（★ 最多报到一下「我起来了」）。
use std::sync::mpsc;
use std::time::Duration;

use aios_worldcore::core::command_bridge::CommandRoute;

const MAP: &str = "/root/aios/crates/aios-ontology/data/command-map.jsonl";
const REGISTER: &str = "world/register";
const REPLY: &str = "world/register/result";

fn as_text(z: &zenoh::bytes::ZBytes) -> String {
    String::from_utf8_lossy(&z.to_bytes()).into_owned()
}

fn main() {
    tokio::runtime::Builder::new_multi_thread()
        .enable_all()
        .build()
        .unwrap()
        .block_on(run());
}

async fn run() {
    let session = zenoh::open(zenoh::Config::from_env().expect("cfg"))
        .await
        .expect("open");
    let (tx, rx) = mpsc::channel::<String>();
    let sub = session.declare_subscriber(REPLY).await.expect("sub");
    tokio::spawn(async move {
        while let Ok(s) = sub.recv_async().await {
            let _ = tx.send(as_text(s.payload()));
        }
    });
    let map = std::fs::read_to_string(MAP).expect("read map");
    let mut sent = 0usize;
    let mut ok = 0usize;
    let mut done: Vec<String> = Vec::new();
    for line in map.lines() {
        if line.trim().is_empty() {
            continue;
        }
        let route = match CommandRoute::parse(line) {
            Ok(r) => r,
            Err(e) => {
                println!("[安装] 表里有坏行，跳过：{e}");
                continue;
            }
        };
        sent += 1;
        let _ = session.put(REGISTER, line).await;
        match rx.recv_timeout(Duration::from_secs(2)) {
            Ok(body) => {
                let v: serde_json::Value = serde_json::from_str(&body).unwrap_or_default();
                if v["ok"].as_bool().unwrap_or(false) {
                    ok += 1;
                    done.push(route.action.clone());
                } else {
                    println!("[安装] {} 被本体拒：{}", route.action, v["why"]);
                }
            }
            Err(_) => {
                println!("[安装] ★ 本体没回话 ⇒ 装不上（★ 本体要先装好 —— 这是那条先后顺序）");
                break;
            }
        }
    }
    println!("[安装] 注册完成：发出 {sent} 条，本体收下 {ok} 条");
    println!("[安装] ★ 这些名字从此【一直留在本体里】，★ 启动时不必再注册（卸载时才删）");
    if !done.is_empty() {
        println!("[安装] 已登记：{}", done.join(" "));
    }
}

```

## 六 · executor_daemon.rs（★ 启动只报到）（逐字）

```rust
//! 独立执行体（T-256）：★ 那只手 —— 坐在总线的另一头。
//!
//! ★★★ 本笔改了注册那一步（★ 上一版是它自己直接写注册表文件 ⇒ 那是绕过本体）：
//!   ① 启动时【不注册】—— ★ 只**报到**一声「我起来了」（world/alive）；
//!      ★★ 注册在【安装时】已经做过了（见 install_actions）；★ 本体里一直留着；
//!      ★ 「在册 ≠ 活着」：★ 报到只改「活着」那一格。
//!   ② 订阅 world/exec，收本体发来的 {action, params, by, trace};
//!   ③ 查映射表 ⇒ program+args ⇒ 真起进程（不拼 shell）;
//!   ④ 回结果到 world/exec/result，并把这一笔落账（只追加）。
use std::time::Duration;

use aios_worldcore::core::command_bridge::CommandRoute;
use aios_worldcore::core::types::Who;
use aios_worldcore::engine::execute::{CommandRunner, Executor};
use aios_worldcore::engine::ledger::Ledger;

const ONT: &str = "/root/aios/crates/aios-ontology/data/ontology.json";
const MAP: &str = "/root/aios/crates/aios-ontology/data/command-map.jsonl";
const LEDGER: &str = "/root/bus/ledger.jsonl";
const ALIVE: &str = "world/alive";
const INTENT: &str = "world/exec";
const RESULT: &str = "world/exec/result";

fn as_text(z: &zenoh::bytes::ZBytes) -> String {
    String::from_utf8_lossy(&z.to_bytes()).into_owned()
}

fn main() {
    tokio::runtime::Builder::new_multi_thread()
        .enable_all()
        .build()
        .unwrap()
        .block_on(run());
}

async fn run() {
    let session = zenoh::open(zenoh::Config::from_env().expect("cfg"))
        .await
        .expect("open");

    // ① ★★ 启动【不注册】—— 只报到（★ 注册在安装时做过了；★ 名字一直留在本体里）
    let map = std::fs::read_to_string(MAP).expect("read map");
    let mut alive = 0usize;
    for line in map.lines() {
        if line.trim().is_empty() {
            continue;
        }
        if let Ok(route) = CommandRoute::parse(line) {
            let msg = serde_json::json!({"action": route.action, "by": "executor"});
            let _ = session.put(ALIVE, msg.to_string()).await;
            alive += 1;
        }
    }
    println!("[执行体] ① 报到完成：{alive} 条（★ 启动【不注册】—— 注册在安装时做过，名字一直留在本体里）");

    // ② 订阅执行请求
    let runner = CommandRunner::from_map(MAP).expect("build runner");
    let text = std::fs::read_to_string(ONT).expect("ont");
    let sub = session.declare_subscriber(INTENT).await.expect("sub");
    println!("[执行体] ② 起来了，订阅 {INTENT}");

    loop {
        match tokio::time::timeout(Duration::from_secs(8), sub.recv_async()).await {
            Ok(Ok(s)) => {
                let txt = as_text(s.payload());
                println!("[执行体] 收到：{txt}");
                let v: serde_json::Value = match serde_json::from_str(&txt) {
                    Ok(v) => v,
                    Err(e) => {
                        println!("[执行体] 拒绝：不是合法 JSON（{e}）");
                        continue;
                    }
                };
                let action = v["action"].as_str().unwrap_or("").to_string();
                let who_name = v["by"].as_str().unwrap_or("world://unknown").to_string();
                let trace = v["trace"].as_str().unwrap_or("trace-0").to_string();
                let mut params = serde_json::Map::new();
                if let Some(o) = v["params"].as_object() {
                    params = o.clone();
                }
                let who = Who::new(who_name.clone());
                let outcome = runner.run(&action, &who, &params);
                let (okc, body) = match outcome {
                    Ok(out) => (true, out.chars().take(200).collect::<String>()),
                    Err(e) => (false, format!("{e}")),
                };
                println!("[执行体] ③ 跑 {action} ⇒ ok={okc}；body 头：{}", body.replace('\n', " ⏎ "));
                let reply = serde_json::json!({
                    "trace": trace, "action": action, "ok": okc, "body": body
                });
                let _ = session.put(RESULT, reply.to_string()).await;
                println!("[执行体] ④ 已回 {RESULT}");
                if let Ok(mut l) = Ledger::load_file(&text, LEDGER) {
                    let mut p = serde_json::Map::new();
                    p.insert("action".to_string(), serde_json::json!(action));
                    p.insert("trace".to_string(), serde_json::json!(trace));
                    p.insert("ok".to_string(), serde_json::json!(okc));
                    match l.append_persisting(LEDGER, &who, "act", format!("exec:{action}"), p) {
                        Ok(seq) => println!("[执行体] ⑤ 落账：第 {} 笔", seq.0),
                        Err(e) => println!("[执行体] ⑤ 落账被拒：{e}"),
                    }
                }
            }
            Ok(Err(e)) => println!("[执行体] 订阅出错：{e}"),
            Err(_) => {
                println!("[执行体] 8 秒没有新 action，退出");
                break;
            }
        }
    }
}
```

## 六 · ontology_daemon.rs（★ 本体进程，两件事分开收）（逐字）

```rust
//! 本体进程（T-258）：★ 注册的**唯一权威** —— 而且它【分得清两件事】：
//!
//! ★★★ 作者口径（逐字整理）：
//!   「**它不需要每次在启动的时候往本体里头注册，是在安装的时候一次性注册就可以了。
//!     然后这个本体里头一直会有相关的内容，卸载的时候在删除**」。
//!
//! ★ 所以本进程收两条路（★ 两件事，别混）：
//!   · `world/register` —— ★★ **安装时**来的登记：★ 校验 ⇒ 登记（**落盘**）⇒ 回话；
//!     ★ 登记过就一直留着 —— ★ 启动时**不再来**。
//!   · `world/alive` —— ★ **启动时**来的报到：★ 只改「活着」那一格（★ 在册 ≠ 活着）；
//!     ★ 不落盘（★ 活着是瞬时状态，重启后重新报到）。
//!
//! ★ ★ 卸载（把名字删掉）那一条**未做** —— ★ 因为 `Register::retire` 要【系统签发的身份 ＋ 人】，
//!   而在一支 bin 里造那种身份是不对的（★ 那属于「授权」那条线）⇒ ★ 登记为待办。
use std::sync::{Arc, Mutex};
use std::time::Duration;

use aios_worldcore::core::command_bridge::CommandRoute;
use aios_worldcore::core::registration::{Register, Registration};

const REG: &str = "/root/bus/registry.jsonl";
const ASK: &str = "world/register";
const REPLY: &str = "world/register/result";
const ALIVE: &str = "world/alive";

fn as_text(z: &zenoh::bytes::ZBytes) -> String {
    String::from_utf8_lossy(&z.to_bytes()).into_owned()
}

fn main() {
    tokio::runtime::Builder::new_multi_thread()
        .enable_all()
        .build()
        .unwrap()
        .block_on(run());
}

async fn run() {
    let session = zenoh::open(zenoh::Config::from_env().expect("cfg"))
        .await
        .expect("open");
    let book = Arc::new(Mutex::new(Register::load(REG).expect("load registry")));
    {
        let b = book.lock().unwrap();
        println!("[本体进程] ① 起来了；手上已登记 {} 条（★ 从 {REG} 读回 —— 装过的一直在）", b.actions().len());
    }

    // ★ 路一：安装时的登记
    {
        let session = session.clone();
        let book = Arc::clone(&book);
        let sub = session.declare_subscriber(ASK).await.expect("sub");
        println!("[本体进程] ② 订阅 {ASK}（★ 安装时的登记走这条）＋ {ALIVE}（★ 启动时的报到走那条）");
        tokio::spawn(async move {
            while let Ok(s) = sub.recv_async().await {
                let txt = as_text(s.payload());
                let route = CommandRoute::parse(&txt);
                let (ok, action, why) = match route {
                    Err(e) => (false, "<解析失败>".to_string(), format!("{e}")),
                    Ok(r) => match r.check() {
                        Err(e) => (false, r.action.clone(), format!("{e}")),
                        Ok(()) => {
                            let reg: Registration = r.as_registration();
                            let a = reg.action.clone();
                            let mut b = book.lock().unwrap();
                            match b.register(REG, reg) {
                                Ok(()) => (true, a, "登记了（★ 落盘，从此一直留着）".to_string()),
                                Err(e) => (false, a, format!("{e}")),
                            }
                        }
                    },
                };
                let n = book.lock().unwrap().actions().len();
                println!("[本体进程] ③ 登记 {action} ⇒ ok={ok}；{why}（★ 现在在册 {n} 条）");
                let reply = serde_json::json!({"ok": ok, "action": action, "why": why});
                let _ = session.put(REPLY, reply.to_string()).await;
            }
        });
    }

    // ★ 路二：启动时的报到（★ 不落盘、不改在册，只改「活着」）
    let alive_sub = session.declare_subscriber(ALIVE).await.expect("sub");
    loop {
        match tokio::time::timeout(Duration::from_secs(10), alive_sub.recv_async()).await {
            Ok(Ok(s)) => {
                let txt = as_text(s.payload());
                let v: serde_json::Value = serde_json::from_str(&txt).unwrap_or_default();
                let action = v["action"].as_str().unwrap_or("").to_string();
                let mut b = book.lock().unwrap();
                let known = b.get(&action).is_some();
                if known {
                    let _ = b.report_alive(&action);
                }
                println!(
                    "[本体进程] ④ 报到 {action}（★ 启动时不注册）⇒ {}",
                    if known { "在册 ✓ 标成活" } else { "★ 不在册 ⇒ 它没装过（★ 登记只在安装时发生）" }
                );
            }
            Ok(Err(e)) => println!("[本体进程] 订阅出错：{e}"),
            Err(_) => {
                println!("[本体进程] 10 秒没有消息，退出");
                break;
            }
        }
    }
}
```
